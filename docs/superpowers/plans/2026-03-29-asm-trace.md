#!/usr/bin/env python3
"""
SMRPG ROM Reader — 65816 disassembler and ROM analysis tool.
Hardcoded for Super Mario RPG (LoROM + SA-1).
"""

import sys
import struct
import argparse
from typing import Union

ROM_PATH = "/mnt/d/smrpg.sfc"


def parse_address(addr: Union[str, int]) -> int:
    """Parse an address from various formats into a raw integer.
    Accepts: 0xC08000, 'C0/8000', '$C08000', 'C08000', plain int.
    """
    if isinstance(addr, str):
        addr = addr.strip().lstrip("$")
        addr = addr.replace("/", "").replace(":", "")
        return int(addr, 16)
    return addr


def snes_to_rom(addr: Union[str, int]) -> int:
    """Convert a SNES address or raw ROM offset to a ROM file offset.

    LoROM mapping for SMRPG:
    - Banks $00-$3F, $80-$BF: offset = (bank & 0x7F) * 0x8000 + (addr & 0x7FFF), addr must be >= $8000
    - Banks $C0-$FF: offset = (bank & 0x3F) * 0x8000 + (addr & 0x7FFF), addr must be >= $8000
    - Values < $008000 that aren't in a bank context: treated as raw ROM offsets
    """
    addr = parse_address(addr)

    # If it looks like a SNES address (has a bank byte >= $00 with addr >= $8000,
    # or bank >= $C0), do the mapping
    bank = (addr >> 16) & 0xFF
    offset = addr & 0xFFFF

    if addr >= 0xC00000:
        # Banks $C0-$FF
        if offset < 0x8000:
            raise ValueError(
                f"SNES address ${addr:06X}: offset ${offset:04X} < $8000 is not ROM in LoROM"
            )
        return (bank & 0x3F) * 0x8000 + (offset - 0x8000)
    elif addr >= 0x800000:
        # Banks $80-$BF (mirror of $00-$3F)
        if offset < 0x8000:
            raise ValueError(
                f"SNES address ${addr:06X}: offset ${offset:04X} < $8000 is not ROM in LoROM"
            )
        return (bank & 0x7F) * 0x8000 + (offset - 0x8000)
    elif bank > 0x00 and offset >= 0x8000:
        # Banks $00-$3F/$40-$7F with addr >= $8000 — LoROM ROM access
        return (bank & 0x3F) * 0x8000 + (offset - 0x8000)
    elif bank == 0x00 and offset >= 0x8000:
        # Bank $00 with addr >= $8000
        return offset - 0x8000
    else:
        # Treat as raw ROM offset
        return addr


def rom_to_snes(offset: int) -> int:
    """Convert a ROM file offset to a SNES address (bank $C0+)."""
    bank = 0xC0 + (offset // 0x8000)
    addr = 0x8000 + (offset % 0x8000)
    return (bank << 16) | addr


class RomReader:
    """Reads and analyzes an SMRPG ROM file."""

    def __init__(self, rom_path: str = ROM_PATH):
        with open(rom_path, "rb") as f:
            self.data = f.read()
        # Detect and skip SMC header (512 bytes if file size % 0x8000 == 512)
        if len(self.data) % 0x8000 == 512:
            self.data = self.data[512:]
        self.size = len(self.data)

    def read_bytes(self, rom_offset: int, length: int = 16) -> bytes:
        """Read raw bytes at a ROM file offset."""
        if rom_offset < 0 or rom_offset >= self.size:
            raise ValueError(f"ROM offset {rom_offset:#x} out of range (ROM size: {self.size:#x})")
        end = min(rom_offset + length, self.size)
        return self.data[rom_offset:end]

    def read_at(self, addr: Union[str, int], length: int = 16) -> bytes:
        """Read bytes at a SNES address or ROM offset."""
        return self.read_bytes(snes_to_rom(addr), length)

    def hex_dump(self, addr: Union[str, int], length: int = 16) -> str:
        """Produce a hex dump at a SNES address."""
        rom_offset = snes_to_rom(addr)
        data = self.read_bytes(rom_offset, length)
        lines = []
        for i in range(0, len(data), 16):
            chunk = data[i:i + 16]
            snes = rom_to_snes(rom_offset + i)
            bank = (snes >> 16) & 0xFF
            off = snes & 0xFFFF
            hex_part = " ".join(f"{b:02X}" for b in chunk)
            ascii_part = "".join(chr(b) if 0x20 <= b < 0x7F else "." for b in chunk)
            lines.append(f"{bank:02X}/{off:04X} | {hex_part:<48s} | {ascii_part}")
        return "\n".join(lines)

# Instructions that are branches (affect control flow)
BRANCH_MNEMONICS = {"BPL", "BMI", "BVC", "BVS", "BCC", "BCS", "BNE", "BEQ", "BRA"}
JUMP_MNEMONICS = {"JMP", "JML", "JSR", "JSL"}
RETURN_MNEMONICS = {"RTS", "RTL", "RTI"}


@dataclass
class Instruction:
    address: int          # SNES address
    opcode: int           # Raw opcode byte
    mnemonic: str         # e.g. "LDA"
    addr_mode: str        # e.g. "imm_m"
    raw_bytes: bytes      # All bytes of the instruction
    length: int           # Total instruction length
    operand_str: str = "" # Formatted operand string
    branch_target: int = 0  # Target address for branches/jumps (0 if N/A)
    sets_m: bool = False  # SEP sets M flag
    sets_x: bool = False  # SEP sets X flag
    clears_m: bool = False  # REP clears M flag
    clears_x: bool = False  # REP clears X flag
    is_branch: bool = False
    is_jump: bool = False
    is_return: bool = False

    def __str__(self) -> str:
        bank = (self.address >> 16) & 0xFF
        off = self.address & 0xFFFF
        hex_str = " ".join(f"{b:02X}" for b in self.raw_bytes)
        asm = f"{self.mnemonic} {self.operand_str}".strip()
        return f"{bank:02X}/{off:04X}: {hex_str:<12s}  {asm}"


def disassemble_instruction(data: bytes, address: int, m_flag: bool = True, x_flag: bool = True) -> Instruction:
    """Disassemble a single 65816 instruction.

    Args:
        data: Bytes starting at the instruction
        address: SNES address of this instruction (for branch target calculation)
        m_flag: True = 8-bit accumulator, False = 16-bit
        x_flag: True = 8-bit index registers, False = 16-bit
    """
    if isinstance(address, str):
        address = parse_address(address)

    opcode = data[0]
    mnemonic, addr_mode = OPCODES[opcode]

    # Calculate instruction length
    base_len = ADDR_MODE_LENGTHS[addr_mode]
    length = base_len
    if addr_mode == IMM_M and not m_flag:
        length += 1
    elif addr_mode == IMM_X and not x_flag:
        length += 1

    raw = data[:length]

    # Format operand string and calculate branch targets
    operand_str = ""
    branch_target = 0
    is_branch = mnemonic in BRANCH_MNEMONICS
    is_jump = mnemonic in JUMP_MNEMONICS
    is_return = mnemonic in RETURN_MNEMONICS

    if addr_mode == IMP or addr_mode == ACC:
        operand_str = "" if addr_mode == IMP else "A"
    elif addr_mode in (IMM, IMM_M, IMM_X):
        if length == 2:
            operand_str = f"#${data[1]:02X}"
        else:
            val = data[1] | (data[2] << 8)
            operand_str = f"#${val:04X}"
    elif addr_mode == DP:
        operand_str = f"${data[1]:02X}"
    elif addr_mode == DPX:
        operand_str = f"${data[1]:02X},X"
    elif addr_mode == DPY:
        operand_str = f"${data[1]:02X},Y"
    elif addr_mode == IDP:
        operand_str = f"(${data[1]:02X})"
    elif addr_mode == IDPX:
        operand_str = f"(${data[1]:02X},X)"
    elif addr_mode == IDPY:
        operand_str = f"(${data[1]:02X}),Y"
    elif addr_mode == IDPL:
        operand_str = f"[${data[1]:02X}]"
    elif addr_mode == IDPLY:
        operand_str = f"[${data[1]:02X}],Y"
    elif addr_mode == ABS:
        val = data[1] | (data[2] << 8)
        operand_str = f"${val:04X}"
        if is_jump or mnemonic == "JSR":
            branch_target = ((address >> 16) << 16) | val
    elif addr_mode == ABX:
        val = data[1] | (data[2] << 8)
        operand_str = f"${val:04X},X"
    elif addr_mode == ABY:
        val = data[1] | (data[2] << 8)
        operand_str = f"${val:04X},Y"
    elif addr_mode == ABL:
        val = data[1] | (data[2] << 8) | (data[3] << 16)
        operand_str = f"${val:06X}"
        if is_jump or mnemonic in ("JSL", "JML"):
            branch_target = val
    elif addr_mode == ABLX:
        val = data[1] | (data[2] << 8) | (data[3] << 16)
        operand_str = f"${val:06X},X"
    elif addr_mode == IND:
        val = data[1] | (data[2] << 8)
        operand_str = f"(${val:04X})"
    elif addr_mode == INDX:
        val = data[1] | (data[2] << 8)
        operand_str = f"(${val:04X},X)"
    elif addr_mode == INDL:
        val = data[1] | (data[2] << 8)
        operand_str = f"[${val:04X}]"
    elif addr_mode == REL:
        offset = data[1]
        if offset >= 0x80:
            offset -= 0x100
        branch_target = (address + 2 + offset) & 0xFFFFFF
        operand_str = f"${branch_target:04X}"
    elif addr_mode == RELL:
        offset = data[1] | (data[2] << 8)
        if offset >= 0x8000:
            offset -= 0x10000
        branch_target = (address + 3 + offset) & 0xFFFFFF
        operand_str = f"${branch_target:06X}"
    elif addr_mode == STK:
        operand_str = f"${data[1]:02X},S"
    elif addr_mode == STKY:
        operand_str = f"(${data[1]:02X},S),Y"
    elif addr_mode == BLK:
        operand_str = f"${data[1]:02X},${data[2]:02X}"

    # Track M/X flag changes from SEP/REP
    sets_m = sets_x = clears_m = clears_x = False
    if mnemonic == "SEP" and length >= 2:
        sets_m = bool(data[1] & 0x20)
        sets_x = bool(data[1] & 0x10)
    elif mnemonic == "REP" and length >= 2:
        clears_m = bool(data[1] & 0x20)
        clears_x = bool(data[1] & 0x10)

    return Instruction(
        address=address, opcode=opcode, mnemonic=mnemonic, addr_mode=addr_mode,
        raw_bytes=raw, length=length, operand_str=operand_str,
        branch_target=branch_target,
        sets_m=sets_m, sets_x=sets_x, clears_m=clears_m, clears_x=clears_x,
        is_branch=is_branch, is_jump=is_jump, is_return=is_return,
    )
```

    def disassemble(self, addr: Union[str, int], count: int = 20,
                    m_flag: bool = True, x_flag: bool = True) -> list:
        """Disassemble `count` instructions starting at a SNES address.
        Tracks M/X flag state through SEP/REP instructions.
        """
        rom_offset = snes_to_rom(addr)
        snes_addr = parse_address(addr) if isinstance(addr, str) else addr
        # If it looks like a raw offset, convert to SNES
        if snes_addr < 0x008000 or (0x008000 <= snes_addr < 0x800000 and (snes_addr & 0xFFFF) < 0x8000):
            snes_addr = rom_to_snes(rom_offset)

        instructions = []
        pos = rom_offset
        current_snes = snes_addr
        m = m_flag
        x = x_flag

        for _ in range(count):
            if pos >= self.size:
                break
            remaining = self.data[pos:pos + 5]  # Max instruction is 4 bytes, grab 5 for safety
            if len(remaining) == 0:
                break
            inst = disassemble_instruction(remaining, current_snes, m_flag=m, x_flag=x)
            instructions.append(inst)

            # Update M/X flags
            if inst.sets_m:
                m = True
            if inst.sets_x:
                x = True
            if inst.clears_m:
                m = False
            if inst.clears_x:
                x = False

            pos += inst.length
            current_snes += inst.length

        return instructions

    def disassemble_str(self, addr: Union[str, int], count: int = 20,
                        m_flag: bool = True, x_flag: bool = True) -> str:
        """Disassemble and return formatted string output."""
        instructions = self.disassemble(addr, count, m_flag, x_flag)
        return "\n".join(str(inst) for inst in instructions)

def create_ips_patch(patches: list) -> bytes:
    """Create an IPS patch from a list of (rom_offset, data) tuples.

    IPS format:
    - Header: 'PATCH' (5 bytes)
    - Records: 3-byte offset + 2-byte size + data (repeated)
    - Footer: 'EOF' (3 bytes)
    """
    result = bytearray(b"PATCH")
    for offset, data in patches:
        # 3-byte big-endian offset
        result.append((offset >> 16) & 0xFF)
        result.append((offset >> 8) & 0xFF)
        result.append(offset & 0xFF)
        # 2-byte big-endian size
        size = len(data)
        result.append((size >> 8) & 0xFF)
        result.append(size & 0xFF)
        # Data
        result.extend(data)
    result.extend(b"EOF")
    return bytes(result)

def get_address_info(addr: int) -> str:
    """Look up what ROM region an address falls in."""
    # If it looks like a SNES address, convert first
    if addr >= 0xC00000 or (addr >= 0x008000 and (addr & 0xFFFF) >= 0x8000):
        try:
            addr = snes_to_rom(addr)
        except ValueError:
            pass

    for start, end, desc in ROM_REGIONS:
        if start <= addr <= end:
            snes = rom_to_snes(addr)
            bank = (snes >> 16) & 0xFF
            off = snes & 0xFFFF
            return f"ROM ${addr:06X} (SNES ${bank:02X}/{off:04X}): {desc}"
    return f"ROM ${addr:06X} (SNES ${rom_to_snes(addr) >> 16 & 0xFF:02X}/{rom_to_snes(addr) & 0xFFFF:04X}): Unknown region"

# 65816 Quick Reference

> Source: https://wiki.superfamicom.org/65816-reference

## Flag-Sensitive Instructions
- M flag (bit 5 of P): controls accumulator width. M=1 → 8-bit, M=0 → 16-bit
  - Affects: LDA/STA/ADC/SBC/AND/ORA/EOR/CMP/BIT immediate operand size
- X flag (bit 4 of P): controls index register width. X=1 → 8-bit, X=0 → 16-bit
  - Affects: LDX/LDY/CPX/CPY immediate operand size
- SEP #$xx: sets bits in P (e.g., SEP #$30 sets both M and X → 8-bit mode)
- REP #$xx: clears bits in P (e.g., REP #$20 clears M → 16-bit accumulator)

## Addressing Modes
[Table of all addressing modes with syntax and byte count]

## Opcode Map
[16x16 grid of all 256 opcodes]

## Branch Instructions
[All conditional and unconditional branches with flag conditions]

## Common Patterns
- `SEP #$20 / LDA #$xx` — load 8-bit value
- `REP #$20 / LDA #$xxxx` — load 16-bit value
- `JSL $xxxxxx` — long subroutine call (cross-bank)

# ASM Trace — SMRPG 65816 Analysis

## Overview

Analyze Super Mario RPG's 65816 assembly code through static ROM analysis, disassembly, reference lookup, and patch generation. Designed for a user who does not know assembly — always explain in plain English alongside any ASM output.

## ROM Reader Tool

Located at `~/.claude/skills/asm-trace/rom_reader.py`. Invoke via Bash:

```bash
# Hex dump
python3 ~/.claude/skills/asm-trace/rom_reader.py read <address> [--length N]

# Disassemble
python3 ~/.claude/skills/asm-trace/rom_reader.py disasm <address> [--count N] [--m16] [--x16]

# Search for bytes
python3 ~/.claude/skills/asm-trace/rom_reader.py find "<hex pattern>" [--start ADDR] [--end ADDR]

# Generate IPS patch
python3 ~/.claude/skills/asm-trace/rom_reader.py write-patch <address> "<hex bytes>" -o <output.ips>

# Identify ROM region
python3 ~/.claude/skills/asm-trace/rom_reader.py info <address>
```

**Address formats accepted:** `C0/8000`, `$C08000`, `C08000`, `0xC08000`, `0x010000` (raw ROM offset).

**Default M/X flags:** 8-bit (M=1, X=1). Use `--m16` or `--x16` to start in 16-bit mode. The disassembler automatically tracks SEP/REP instructions to update flag state.

## Operations

### 1. Read (hex dump)
Dump raw bytes. Use when you need to see data tables, check values, or verify patches.

### 2. Disassemble
Decode 65816 instructions. Always start by disassembling the target address to understand the code before making any analysis.

### 3. Trace (manual)
Follow execution flow by disassembling at an entry point, then following JSR/JSL/JMP targets:
1. Disassemble at the entry point
2. Identify subroutine calls (JSR/JSL) and jumps (JMP/JML)
3. Disassemble each target to understand what it does
4. Follow branches (BNE/BEQ/etc.) to understand conditional logic
5. Build up a call graph with plain-English annotations

### 4. Lookup
Consult the reference files to find known addresses, data structures, or functions:
1. Check `INDEX.md` for the topic
2. Load the referenced summary file from `ref/`
3. If the summary is insufficient, read the original source file

### 5. Patch
Generate ROM patches:
1. Identify the exact bytes to change (disassemble first to confirm)
2. Write the new instruction bytes
3. Use `write-patch` to generate an IPS file
4. Explain what the patch does and any risks

### 6. Explain
Given an address range, produce a human-readable explanation:
1. Disassemble the code
2. Annotate each instruction with what it does
3. Identify RAM addresses being read/written (look up in RAM map)
4. Summarize the overall purpose of the routine

## Debugger Escalation

When static analysis cannot resolve something (indirect jumps, runtime register state, self-modifying code, need to observe actual game behavior), instruct the user to use **bsnes-plus v05**:

1. Tell them exactly which address to set a breakpoint/watchpoint on and why
2. Specify what to look for (register values, memory contents, execution path)
3. Ask them to paste back the trace log or register dump
4. Interpret the results and continue analysis

Example prompt to user:
> "I can't determine the jump target statically because it uses an indirect jump through a table. Could you:
> 1. Open the ROM in bsnes-plus
> 2. Set a breakpoint at $C2/C71E
> 3. Trigger a Lucky bonus in battle
> 4. When it breaks, paste the register state (especially A and X)
> That will tell me which flower bonus handler it's jumping to."

## Important Notes

- **Always explain in plain English.** The user doesn't know assembly. Show the ASM but explain what each part does.
- **Verify before patching.** Always disassemble the target area first to confirm you're modifying the right code.
- **SMRPG uses SA-1.** Some code runs on the SA-1 coprocessor, not the main CPU. Be aware of bank mapping differences.
- **LoROM addressing.** The ROM reader handles SNES ↔ ROM offset conversion automatically.
- **The ROM file is vanilla SMRPG** at `/mnt/d/smrpg.sfc`. When analyzing randomizer-specific behavior, note that the randomizer expands the ROM beyond 4MB.
